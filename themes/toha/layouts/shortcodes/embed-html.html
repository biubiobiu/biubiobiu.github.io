{{- $src := .Get "src" -}} {{- if not $src -}} {{- errorf "embed-html shortcode requires a src parameter: %s" .Position -}} {{- end -}} {{- $title := .Get "title" | default "嵌入式文章内容" -}} {{- $source := readFile $src -}} {{- $frameID := printf "embedded-html-%s-%d" (md5 $src) .Ordinal -}} {{/* Let links inside srcdoc navigate the whole page instead of nesting a post in the iframe. */}} {{- $source = replace $source "" "" -}} {{/* Optionally inline a companion script. Files rendered through srcdoc cannot reliably load a sibling script from the repository, so the post can pass script="path/to/annotator.js". */}} {{- $scriptPath := .Get "script" -}} {{- if $scriptPath -}} {{- $scriptSource := readFile $scriptPath -}} {{- $scriptSource = replace $scriptSource "" "<\\/script>" -}} {{- $source = replace $source "" "" -}} {{- $inlineScript := printf "" $scriptSource -}} {{- $source = replace $source "" (printf "%s" $inlineScript) -}} {{- end -}}
